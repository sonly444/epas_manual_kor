# FETCH BULK COLLECT — EPAS 18 SPL

> 원문: [EPAS v18 — SPL › Working with collections › Using the BULK COLLECT clause › FETCH BULK COLLECT](https://www.enterprisedb.com/docs/epas/latest/application_programming/epas_compat_spl/12_working_with_collections/04_using_the_bulk_collect_clause/02_fetch_bulk_collect/)
> EPAS v18 공식 문서의 한글 **전문 번역**입니다. 명령·코드 이름은 원문 그대로 보존했습니다.

`FETCH` 문과 함께 `BULK COLLECT` 절을 사용할 수 있습니다. `FETCH BULK COLLECT`는 결과 집합에서 한 번에 한 행씩 반환하는 대신, `LIMIT` 절로 제한되지 않는 한 결과 집합의 모든 행을 하나 이상의 지정된 컬렉션(스칼라 및 복합 타입 모두)으로 한 번에 반환합니다:

```sql
FETCH <name> BULK COLLECT INTO <collection> [, ...] [ LIMIT <n> ];
```

`FETCH` 문에 대한 자세한 내용은 [커서에서 행 페치(Fetching rows from a cursor)](https://www.enterprisedb.com/docs/epas/latest/application_programming/epas_compat_spl/08_static_cursors/03_fetching_rows_from_a_cursor/#fetching_rows_from_a_cursor)를 참고하십시오.

단일 컬렉션을 지정하는 경우 `collection`은 단일 필드의 컬렉션이거나 레코드 타입의 컬렉션일 수 있습니다. 둘 이상의 컬렉션을 지정하는 경우 각 `collection`은 단일 필드 또는 레코드 타입으로 구성되어야 합니다. `name`으로 식별되는 커서의 `SELECT` 목록에 있는 표현식은 대상 컬렉션의 모든 필드와 개수, 순서, 타입 호환성이 일치해야 합니다. `LIMIT n`을 지정하면 각 `FETCH`에서 컬렉션으로 반환되는 행 수가 `n`을 초과하지 않습니다.

다음 예제는 `FETCH BULK COLLECT` 문을 사용하여 행을 연관 배열로 조회합니다:

```sql
DECLARE
    TYPE emp_tbl IS TABLE OF emp%ROWTYPE INDEX BY BINARY_INTEGER;
    t_emp           EMP_TBL;
    CURSOR emp_cur IS SELECT * FROM emp;
BEGIN
    OPEN emp_cur;
    FETCH emp_cur BULK COLLECT INTO t_emp;
    CLOSE emp_cur;
    DBMS_OUTPUT.PUT_LINE('EMPNO  ENAME    JOB        HIREDATE    ' ||
        'SAL        ' || 'COMM      DEPTNO');
    DBMS_OUTPUT.PUT_LINE('-----  -------  ---------  ---------   ' ||
        '--------   ' || '--------  ------');
    FOR i IN 1..t_emp.COUNT LOOP
        DBMS_OUTPUT.PUT_LINE(t_emp(i).empno || '   ' ||
            RPAD(t_emp(i).ename,8) || ' ' ||
            RPAD(t_emp(i).job,10) || ' ' ||
            TO_CHAR(t_emp(i).hiredate,'DD-MON-YY') || ' ' ||
            TO_CHAR(t_emp(i).sal,'99,999.99') || ' ' ||
            TO_CHAR(NVL(t_emp(i).comm,0),'99,999.99') || '  ' ||
            t_emp(i).deptno);
    END LOOP;
END;
```

```sql
-- 복합 데이터 타입의 오브젝트 타입 두 개를 생성합니다
CREATE TYPE db_type1 as OBJECT(a INT, b VARCHAR2(10));
CREATE TYPE db_type2 as OBJECT(c VARCHAR2(10), d INT);

-- 위 오브젝트 타입을 사용하여 테이블을 생성합니다
CREATE TABLE db_tab(x DB_TYPE1, s1 NUMBER, y DB_TYPE2, s2 TIMESTAMP);

-- 테이블에 행을 삽입합니다
INSERT INTO db_tab values(DB_TYPE1(1, '10'), 1.1, DB_TYPE2('100',1000), '1-Jan-2021 12:30:11 PM');
INSERT INTO db_tab values(DB_TYPE1(2, '20'), 2.2, DB_TYPE2('200',2000), '2-Feb-2022 08:40:52 AM');
INSERT INTO db_tab values(DB_TYPE1(3, '30'), 3.3, DB_TYPE2('300',3000), '3-Jan-2023 04:20:33 PM');
```

```sql
-- 스칼라와 복합 타입을 모두 페치하기 위해 FETCH BULK COLLECT INTO 절을 사용합니다
DECLARE
  TYPE type1_tbl IS TABLE OF db_type1 INDEX BY BINARY_INTEGER;
  TYPE s1_tbl    IS TABLE OF number INDEX BY BINARY_INTEGER;
  TYPE type2_tbl IS TABLE OF db_type2 INDEX BY BINARY_INTEGER;
  TYPE s2_tbl    IS TABLE OF timestamp INDEX BY BINARY_INTEGER;
  x type1_tbl;
  s1 s1_tbl;
  y type2_tbl;
  s2 s2_tbl;
  CURSOR c1 is SELECT * FROM db_tab ORDER BY s1;
BEGIN
  OPEN c1;
  FETCH c1 BULK COLLECT INTO x, s1, y, s2;
  FOR i IN 1..x.count LOOP
    DBMS_OUTPUT.PUT_LINE(x(i)||' '||s1(i)||' '||y(i)||'"'||s2(i)||'"');
  END LOOP;
  CLOSE c1;

  SELECT * BULK COLLECT INTO x, s1, y, s2 FROM db_tab ORDER BY s1;
  FOR i IN 1..x.count LOOP
    DBMS_OUTPUT.PUT_LINE(x(i)||' '||s1(i)||' '||y(i)||'"'||s2(i)||'"');
  END LOOP;
end;
```
